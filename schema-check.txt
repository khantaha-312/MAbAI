generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

model Org {
  id        String   @id @default(cuid())
  name      String
  createdBy String?
  updatedBy String?
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  users      User[]
  portfolios Portfolio[]
}

model User {
  id        String   @id @default(cuid())
  orgId     String
  org       Org      @relation(fields: [orgId], references: [id])
  email     String   @unique
  clerkId   String   @unique
  createdBy String?
  updatedBy String?
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  portfolios Portfolio[]
}

model Instrument {
  id        String   @id @default(cuid())
  symbol    String
  name      String
  assetType String
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  symbolMappings SymbolMapping[]
  positions      Position[]
  riskFlags      RiskFlag[]

  @@unique([symbol, assetType])
}

model SymbolMapping {
  id            String     @id @default(cuid())
  instrumentId  String
  instrument    Instrument @relation(fields: [instrumentId], references: [id])
  provider      String
  providerSymbol String
  createdAt     DateTime   @default(now())
  updatedAt     DateTime   @updatedAt

  @@unique([provider, providerSymbol])
}

model Portfolio {
  id        String    @id @default(cuid())
  orgId     String
  org       Org       @relation(fields: [orgId], references: [id])
  userId    String
  user      User      @relation(fields: [userId], references: [id])
  name      String
  createdBy String?
  updatedBy String?
  deletedAt DateTime?
  createdAt DateTime  @default(now())
  updatedAt DateTime  @updatedAt

  positions Position[]
}

model Position {
  id           String    @id @default(cuid())
  portfolioId  String
  portfolio    Portfolio  @relation(fields: [portfolioId], references: [id])
  instrumentId String
  instrument   Instrument @relation(fields: [instrumentId], references: [id])
  quantity     Decimal
  avgCostBasis Decimal
  createdBy    String?
  updatedBy    String?
  deletedAt    DateTime?
  createdAt    DateTime  @default(now())
  updatedAt    DateTime  @updatedAt

  riskFlags RiskFlag[]
}

model RiskFlag {
  id         String     @id @default(cuid())
  positionId String
  position   Position   @relation(fields: [positionId], references: [id])
  instrumentId String
  instrument Instrument @relation(fields: [instrumentId], references: [id])
  flagType   String
  severity   String
  createdAt  DateTime   @default(now())
  updatedAt  DateTime   @updatedAt
}
